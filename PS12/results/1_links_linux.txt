--- lab1_item3: level 1 and level 2 link chains ---
# Symbolic link chain: src_s <- l1_sym <- l2_sym
$ echo 'chain data' > src_s.txt
$ ln -s src_s.txt l1_sym.txt && ln -s l1_sym.txt l2_sym.txt
$ ls -li src_s.txt l1_sym.txt l2_sym.txt
84 lrwxrwxrwx 1 kiatisakk kiatisakk  9 Nov 20 17:04 l1_sym.txt -> src_s.txt
85 lrwxrwxrwx 1 kiatisakk kiatisakk 10 Nov 20 17:04 l2_sym.txt -> l1_sym.txt
83 -rw-r--r-- 1 kiatisakk kiatisakk 11 Nov 20 17:04 src_s.txt
$ cat l2_sym.txt
chain data
$ rm l1_sym.txt   # delete level 1
$ cat l2_sym.txt
cat: l2_sym.txt: No such file or directory
$ ln -s src_s.txt l1_sym.txt && rm src_s.txt   # restore level 1, delete source
$ cat l1_sym.txt
cat: l1_sym.txt: No such file or directory
$ cat l2_sym.txt
cat: l2_sym.txt: No such file or directory
